postal_code
Compliance
US ZIP must be exactly 5 digits or ZIP+4 (9 digits with hyphen). Validate based on the normalised country column.
US_ZIP   = re.compile(r'^\d{5}(-\d{4})?$')
UK_POST  = re.compile(r'^[A-Z]{1,2}\d[A-Z\d]? ?\d[A-Z]{2}$', re.I)
PL_POST  = re.compile(r'^\d{2}-\d{3}$')

def validate_postal(country, postal):
    postal = str(postal).strip()
    if country == 'USA': return bool(US_ZIP.match(postal))
    if country == 'GBR': return bool(UK_POST.match(postal))
    if country == 'POL': return bool(PL_POST.match(postal))
    return True  # unknown country — skip

CURRENCY_CHARS = re.compile(r'[\$£¥€]|PLN|USD|GBP|CNY', re.I)

def parse_revenue(v):
    v = str(v).strip()
    v = CURRENCY_CHARS.sub('', v).strip()
    return round(float(v), 2)

df['revenue_usd'] = df['revenue'].apply(parse_revenue)
